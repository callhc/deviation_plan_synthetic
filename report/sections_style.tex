\newcommand{\missingassetbox}[1]{%
  \fbox{%
    \parbox[c][0.22\textheight][c]{0.82\linewidth}{%
      \centering\ttfamily Missing asset:\\\detokenize{#1}%
    }%
  }%
}

\newcommand{\safeincludegraphics}[2][]{%
  \IfFileExists{#2}{%
    \includegraphics[#1]{#2}%
  }{%
    \IfFileExists{pictures/#2}{%
      \includegraphics[#1]{#2}%
    }{%
      \IfFileExists{chapters/introduction/pictures/#2}{%
        \includegraphics[#1]{#2}%
      }{%
        \IfFileExists{chapters/methodology/pictures/#2}{%
          \includegraphics[#1]{#2}%
        }{%
          \IfFileExists{../visualisations/#2}{%
            \includegraphics[#1]{#2}%
          }{%
            \missingassetbox{#2}%
          }%
        }%
      }%
    }%
  }%
}
